# MERN Chat App (using `ws` — WebSocket library)

This document contains a full-stack MERN chat application that uses the low-level `ws` WebSocket library for real-time communication. It includes:

* Project folder structure
* Backend (Node.js + Express + Mongoose + `ws`) with JWT auth and message persistence
* Frontend (React) that logs in/registers, connects to the WebSocket server using JWT, sends/receives messages, shows online users and message history
* `package.json` examples and `.env` example

> **Note:** This is a working, opinionated starter. You can expand rooms, typing indicators, file attachments, delivery receipts, etc.

### server/src/routes/auth.js

```javascript
const express = require('express');
const bcrypt = require('bcrypt');
const jwt = require('jsonwebtoken');
const User = require('../models/User');

const router = express.Router();

router.post('/register', async (req, res) => {
  const { username, password } = req.body;
  if (!username || !password) return res.status(400).json({ message: 'Missing fields' });
  try {
    const existing = await User.findOne({ username });
    if (existing) return res.status(409).json({ message: 'Username taken' });
    const passwordHash = await bcrypt.hash(password, 10);
    const user = await User.create({ username, passwordHash });
    const token = jwt.sign({ id: user._id, username: user.username }, process.env.JWT_SECRET);
    res.json({ user: { id: user._id, username: user.username }, token });
  } catch (err) {
    console.error(err);
    res.status(500).json({ message: 'Server error' });
  }
});

router.post('/login', async (req, res) => {
  const { username, password } = req.body;
  if (!username || !password) return res.status(400).json({ message: 'Missing fields' });
  try {
    const user = await User.findOne({ username });
    if (!user) return res.status(401).json({ message: 'Invalid credentials' });
    const ok = await bcrypt.compare(password, user.passwordHash);
    if (!ok) return res.status(401).json({ message: 'Invalid credentials' });
    const token = jwt.sign({ id: user._id, username: user.username }, process.env.JWT_SECRET);
    res.json({ user: { id: user._id, username: user.username }, token });
  } catch (err) {
    console.error(err);
    res.status(500).json({ message: 'Server error' });
  }
});

module.exports = router;
```

### server/src/middleware/auth.js

```javascript
const jwt = require('jsonwebtoken');

function expressAuth(req, res, next) {
  const auth = req.headers.authorization;
  if (!auth) return res.status(401).json({ message: 'No token' });
  const token = auth.split(' ')[1];
  try {
    const data = jwt.verify(token, process.env.JWT_SECRET);
    req.user = data;
    next();
  } catch (err) {
    return res.status(401).json({ message: 'Invalid token' });
  }
}

module.exports = { expressAuth };
```

### server/src/ws-handler.js

```javascript
// Handles ws connections: authenticate using token in query, map sockets to users, broadcast messages, persist messages
const jwt = require('jsonwebtoken');
const url = require('url');
const Message = require('./models/Message');

// Map userId -> ws
const clients = new Map();

function setupWsServer(wss) {
  wss.on('connection', async function connection(ws, req) {
    // Authenticate: token sent as ?token=... in WebSocket URL
    const { query } = url.parse(req.url, true);
    const token = query.token;
    if (!token) {
      ws.send(JSON.stringify({ type: 'error', message: 'No token provided' }));
      ws.close();
      return;
    }

    let user;
    try {
      user = jwt.verify(token, process.env.JWT_SECRET);
    } catch (err) {
      ws.send(JSON.stringify({ type: 'error', message: 'Invalid token' }));
      ws.close();
      return;
    }

    // store
    ws.user = user; // { id, username }
    clients.set(user.id, ws);

    // Send existing last 50 messages
    const recent = await Message.find().sort({ createdAt: 1 }).limit(200).populate('from', 'username');
    ws.send(JSON.stringify({ type: 'history', messages: recent.map(m => ({ id: m._id, text: m.text, from: m.from.username, createdAt: m.createdAt })) }));

    // Notify others about presence
    broadcast({ type: 'presence', user: { id: user.id, username: user.username }, action: 'online' });

    ws.on('message', async (raw) => {
      try {
        const msg = JSON.parse(raw);
        if (msg.type === 'message') {
          // Save to DB
          const saved = await Message.create({ from: user.id, text: msg.text });
          const out = { type: 'message', message: { id: saved._id, from: user.username, text: saved.text, createdAt: saved.createdAt } };
          broadcast(out);
        }
      } catch (err) {
        console.error('ws message error', err);
      }
    });

    ws.on('close', () => {
      clients.delete(user.id);
      broadcast({ type: 'presence', user: { id: user.id, username: user.username }, action: 'offline' });
    });
  });
}

function broadcast(obj) {
  const data = JSON.stringify(obj);
  for (const [, clientWs] of clients) {
    if (clientWs.readyState === clientWs.OPEN) clientWs.send(data);
  }
}

module.exports = { setupWsServer };
```

### client/src/wsClient.js

```javascript
// Lightweight wrapper to create a WebSocket with reconnection attempts
export function createWs(token, onOpen, onMessage, onClose) {
  const url = `${(window.location.protocol === 'https:' ? 'wss' : 'ws')}://` + (process.env.REACT_APP_WS_HOST || 'localhost:4000') + `/ws?token=${encodeURIComponent(token)}`;
  let ws;
  let attempts = 0;
  function connect() {
    ws = new WebSocket(url);
    ws.onopen = (e) => { attempts = 0; onOpen && onOpen(e); };
    ws.onmessage = (e) => { onMessage && onMessage(JSON.parse(e.data)); };
    ws.onclose = (e) => { onClose && onClose(e); setTimeout(() => { attempts++; connect(); }, Math.min(10000, 1000 * attempts)); };
    ws.onerror = (err) => console.error('WS error', err);
  }
  connect();
  return {
    send(obj) { if (ws && ws.readyState === ws.OPEN) ws.send(JSON.stringify(obj)); }
  };
}
```

### client/src/components/Login.js

```javascript
import React, { useState } from 'react';
import { login, register } from '../api';

export default function Login({ onAuth }) {
  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');
  const [mode, setMode] = useState('login');
  const [err, setErr] = useState(null);

  async function submit(e) {
    e.preventDefault();
    setErr(null);
    try {
      const res = mode === 'login' ? await login(username, password) : await register(username, password);
      if (res.token) {
        localStorage.setItem('token', res.token);
        onAuth(res.user, res.token);
      } else {
        setErr(res.message || 'Auth failed');
      }
    } catch (err) { setErr('Network error'); }
  }

  return (
    <div className="login">
      <h2>{mode === 'login' ? 'Login' : 'Register'}</h2>
      <form onSubmit={submit}>
        <input value={username} onChange={e=>setUsername(e.target.value)} placeholder="username" required />
        <input value={password} onChange={e=>setPassword(e.target.value)} placeholder="password" type="password" required />
        <button type="submit">{mode === 'login' ? 'Login' : 'Register'}</button>
      </form>
      <button onClick={()=>setMode(mode === 'login' ? 'register' : 'login')}>Switch to {mode === 'login' ? 'Register' : 'Login'}</button>
      {err && <div className="error">{err}</div>}
    </div>
  );
}
```

### client/src/components/Chat.js

```javascript
import React, { useState, useEffect, useRef } from 'react';
import MessageList from './MessageList';
import { fetchMessages } from '../api';
import { createWs } from '../wsClient';

export default function Chat({ user, token }) {
  const [messages, setMessages] = useState([]);
  const [input, setInput] = useState('');
  const wsRef = useRef(null);

  useEffect(() => {
    let wsClient = createWs(token,
      () => console.log('ws open'),
      (msg) => {
        if (msg.type === 'history') setMessages(msg.messages);
        else if (msg.type === 'message') setMessages(prev => [...prev, msg.message]);
        else if (msg.type === 'presence') console.log('presence', msg);
      },
      () => console.log('ws closed')
    );
    wsRef.current = wsClient;

    // also fetch via REST as fallback
    fetchMessages().then(arr => { if (arr && arr.length) setMessages(arr); });

    return () => { /* no explicit close in this wrapper */ };
  }, [token]);

  function send() {
    if (!input.trim()) return;
    wsRef.current.send({ type: 'message', text: input });
    setInput('');
  }

  return (
    <div className="chat">
      <h3>Welcome, {user.username}</h3>
      <MessageList messages={messages} />
      <div className="composer">
        <input value={input} onChange={e=>setInput(e.target.value)} placeholder="Type a message" />
        <button onClick={send}>Send</button>
      </div>
    </div>
  );
}
```

### client/src/App.js

```javascript
import React, { useState, useEffect } from 'react';
import Login from './components/Login';
import Chat from './components/Chat';

function App(){
  const [user, setUser] = useState(null);
  const [token, setToken] = useState(localStorage.getItem('token'));

  useEffect(()=>{
    if (token && !user) {
      // Optionally decode token to populate username or call an endpoint
      try {
        const payload = JSON.parse(atob(token.split('.')[1]));
        setUser({ id: payload.id, username: payload.username });
      } catch(e){ console.log('failed decode'); }
    }
  }, [token, user]);

  function onAuth(u, t) {
    setUser(u); setToken(t);
  }

  if (!token || !user) return <Login onAuth={onAuth} />;
  return <Chat user={user} token={token} />;
}

export default App;
```

## 3) How to run

1. Install MongoDB and ensure it's running.
2. Backend:

   * `cd server`
   * copy `.env.example` to `.env` and edit values
   * `npm install`
   * `npm run dev` (or `npm start`)
3. Frontend:

   * `cd client`
   * `npm install`
   * `npm start`
4. Open `http://localhost:3000` and register/login. The app will connect to `ws://localhost:4000/ws?token=...`.
